import Card from "@/components/nw/Card";
import Eyebrow from "@/components/nw/Eyebrow";
import Badge from "@/components/nw/Badge";

  | "F1" | "F2" | "F3" | "F4" | "F5" | "F6"

  body: string;
  wave: PlaceholderWave;
  className?: string;

  eyebrow,
  headline,

        {body}
      </p>

}
